页面引用代码

<!-- 1. React 组件内（Vite，文件放 public/ 下按根路径引用） -->
<img src="/logo.svg" width="24" height="24" alt="Zwico" />

<!-- 2. 标题栏 / 关于页组合 -->
<span style="display:inline-flex;align-items:center;gap:8px">
  <img src="/logo.svg" width="20" height="20" alt="" />
  <b>Zwico</b>
</span>

<!-- 3. favicon（index.html） -->
<link rel="icon" type="image/svg+xml" href="/logo.svg" />

<!-- 4. 单色变体：颜色随 CSS currentColor / color 继承 -->
<img class="mono" src="/logo-mono.svg" width="20" height="20" alt="" />
<style>.mono { color: var(--accent); }</style>
